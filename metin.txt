            if(target) target.classList.remove('hidden');

            document.querySelectorAll('.nav-btn').forEach(btn => btn.classList.remove('nav-active'));
            const activeBtn = document.getElementById(`nav-${tabId}`);
            if (activeBtn) activeBtn.classList.add('nav-active');

            if(tabId === 'journal') renderJournal();
            if(tabId === 'e-lessons') renderLessons(lessonsData);
            if(tabId === 'assignments') renderAssignments();

        // ================= 1. JOURNAL ENGINE =================
        function renderJournal() {
            const tbody = document.getElementById('journalTableBody');
            tbody.innerHTML = journalStudents.map((st, idx) => {
                const grades = [st.g1, st.g2, st.g3, parseInt(st.gToday)].filter(g => !isNaN(g) && g > 0);
                const avg = grades.length ? (grades.reduce((a,b)=>a+b,0)/grades.length).toFixed(1) : '-';

                return `
                    <tr class="hover:bg-slate-50 dark:hover:bg-slate-800/40 transition">
                        <td class="p-4 text-center font-bold text-slate-400">${idx + 1}</td>
                        <td class="p-4 font-bold text-slate-800 dark:text-slate-200">${st.name}</td>
                        <td class="p-4 text-center">
                            <button onclick="toggleAttendance(${st.id})" class="px-2.5 py-1 rounded-lg text-xs font-bold ${st.status === 'i/ə' ? 'bg-emerald-500/10 text-emerald-600 border border-emerald-500/30' : 'bg-rose-500/10 text-rose-600 border border-rose-500/30'}">
                                ${st.status === 'i/ə' ? 'İştirak Edir' : 'Qayıb (Q)'}
                            </button>
                        </td>
                        <td class="p-4 text-center font-semibold text-slate-600 dark:text-slate-400">${st.g1}</td>
                        <td class="p-4 text-center font-semibold text-slate-600 dark:text-slate-400">${st.g2}</td>
                        <td class="p-4 text-center font-semibold text-slate-600 dark:text-slate-400">${st.g3}</td>
                        <td class="p-4 text-center">
                            <input type="number" min="1" max="10" value="${st.gToday}" onchange="updateGrade(${st.id}, this.value)" class="w-14 text-center bg-slate-100 dark:bg-slate-800 border border-slate-300 dark:border-slate-700 rounded-lg py-1 font-bold text-slate-800 dark:text-white outline-none focus:ring-2 focus:ring-brand-500">
                        </td>
                        <td class="p-4 text-center font-bold text-brand-600 text-base">${avg}</td>
                    </tr>
                `;
            }).join('');
        }

        function toggleAttendance(id) {
            const student = journalStudents.find(s => s.id === id);
            if(student) {
                student.status = student.status === 'i/ə' ? 'q' : 'i/ə';
                renderJournal();
            }
        }

        function updateGrade(id, val) {
            const student = journalStudents.find(s => s.id === id);
            if(student) {
                student.gToday = val;
                renderJournal();
            }
        }

        function openLessonModal(id) {
            const lesson = lessonsData.find(l => l.id === id);
            if(!lesson) return;

            currentLessonSlides = lesson.slides;
            currentSlideIdx = 0;

            document.getElementById('modalLessonSubject').innerText = lesson.subjectName;
            document.getElementById('modalLessonTitle').innerText = lesson.title;
            
            updateSlideView();
            document.getElementById('lessonViewerModal').classList.remove('hidden');
        }

        function updateSlideView() {
            document.getElementById('modalLessonContent').innerHTML = `
                <div class="space-y-4">
                    <span class="text-xs font-bold text-brand-600">Slayd İzahı #${currentSlideIdx + 1}</span>
                    <h4 class="fredoka text-lg font-bold text-slate-800 dark:text-white leading-snug">${currentLessonSlides[currentSlideIdx]}</h4>
                </div>
                <div class="pt-4 text-xs text-slate-400">Şagirdlərin diqqətinə: Slaydı oxuduqdan sonra növbəti düyməsini sıxın.</div>
            `;
            document.getElementById('slideIndicator').innerText = `Slayd ${currentSlideIdx + 1} / ${currentLessonSlides.length}`;
            document.getElementById('prevSlideBtn').disabled = currentSlideIdx === 0;
            document.getElementById('nextSlideBtn').disabled = currentSlideIdx === currentLessonSlides.length - 1;
        }

        function changeSlide(dir) {

        let quizState = {
            category: 'az',
            currentIndex: 0,
            questions: [],
            selectedOption: null,
            score: 0,
            timer: null,
            timeLeft: 180
        };

        function initQuiz(cat) {
            quizState.category = cat;
            quizState.questions = quizData[cat];
            quizState.currentIndex = 0;
            quizState.score = 0;
            quizState.selectedOption = null;
            quizState.timeLeft = 180;

            document.getElementById('quiz-cat-az').className = cat === 'az' ? 'px-4 py-2 rounded-xl text-xs font-bold bg-brand-600 text-white shadow' : 'px-4 py-2 rounded-xl text-xs font-bold bg-slate-200 dark:bg-slate-800 text-slate-700 dark:text-slate-300';
            document.getElementById('quiz-cat-math').className = cat === 'math' ? 'px-4 py-2 rounded-xl text-xs font-bold bg-brand-600 text-white shadow' : 'px-4 py-2 rounded-xl text-xs font-bold bg-slate-200 dark:bg-slate-800 text-slate-700 dark:text-slate-300';

            document.getElementById('quizContainer').classList.remove('hidden');
            document.getElementById('quizSummary').classList.add('hidden');

            startTimer();
            renderQuestion();
        }

        function startTimer() {
            clearInterval(quizState.timer);
            quizState.timer = setInterval(() => {
                quizState.timeLeft--;
                const min = Math.floor(quizState.timeLeft / 60).toString().padStart(2, '0');
                const sec = (quizState.timeLeft % 60).toString().padStart(2, '0');
                document.getElementById('quizTimer').innerText = `${min}:${sec}`;

                if(quizState.timeLeft <= 0) {
                    clearInterval(quizState.timer);
                    finishQuiz();
                }
            }, 1000);
        }

        function renderQuestion() {
            const q = quizState.questions[quizState.currentIndex];
            document.getElementById('quizProgress').innerText = `Sual: ${quizState.currentIndex + 1} / ${quizState.questions.length}`;
            document.getElementById('quizQuestion').innerText = q.q;

            const optionsContainer = document.getElementById('quizOptions');
            optionsContainer.innerHTML = q.options.map((opt, idx) => `
                <button onclick="selectOption(${idx})" class="quiz-opt-btn text-left p-4 rounded-2xl border border-slate-200 dark:border-slate-700 font-medium text-sm hover:border-brand-500 transition flex items-center justify-between">
                    <span>${opt}</span>
                    <i class="fa-regular fa-circle text-slate-300"></i>
                </button>
            `).join('');

            document.getElementById('nextQuestionBtn').disabled = true;
            quizState.selectedOption = null;
        }

        function selectOption(idx) {
            quizState.selectedOption = idx;
            const btns = document.querySelectorAll('.quiz-opt-btn');
            btns.forEach((b, i) => {
                if(i === idx) {
                    b.className = "quiz-opt-btn text-left p-4 rounded-2xl border-2 border-brand-600 bg-brand-500/10 font-bold text-sm text-brand-600 flex items-center justify-between shadow-sm";
                } else {
                    b.className = "quiz-opt-btn text-left p-4 rounded-2xl border border-slate-200 dark:border-slate-700 font-medium text-sm opacity-60 flex items-center justify-between";
                }
            });
            document.getElementById('nextQuestionBtn').disabled = false;
        }

        function nextQuestion() {
            const q = quizState.questions[quizState.currentIndex];
            if(quizState.selectedOption === q.correct) {
                quizState.score++;
            }

            quizState.currentIndex++;
            if(quizState.currentIndex < quizState.questions.length) {
                renderQuestion();
            } else {
                finishQuiz();
            }
        }

        function finishQuiz() {
            clearInterval(quizState.timer);
            document.getElementById('quizContainer').classList.add('hidden');
            document.getElementById('quizSummary').classList.remove('hidden');

            const total = quizState.questions.length;
            const percent = Math.round((quizState.score / total) * 100);
            const starsEarned = quizState.score * 10;

            document.getElementById('summaryCorrect').innerText = `${quizState.score} / ${total}`;
            document.getElementById('summaryPercent').innerText = `${percent}%`;
            document.getElementById('summaryStars').innerText = `+${starsEarned} ★`;

            if (typeof confetti === 'function') {
                confetti({ particleCount: 50, spread: 70, origin: { y: 0.7 } });
            }
        }
